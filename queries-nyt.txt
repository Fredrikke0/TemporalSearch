# What date did Fidel Castro win the Cuban revolution?
SELECT date, COUNT(*) FROM source WHERE CONTAINS('Cuban') AND CONTAINS('victory') AND NER(PERSON, 'Fidel Castro') AND NER(PERSON, 'Fulgencio Batista') AND DATE(< 2050) BIND date GROUP BY date ORDER BY COUNT(*) DESC ::: 1959-01-01

# List of presidents of the us with order.
SELECT ordinal, pres, COUNT(*) FROM source WHERE CONTAINS('president') AND CONTAINS('united states') AND NER(ORDINAL) BIND ordinal AND NER(PERSON) BIND pres GROUP BY ordinal, pres GRANULARITY SENTENCE ORDER BY COUNT(*) DESC LIMIT 10
# List of presidents after president.

# List of unemployment rates
SELECT country, rate, date FROM articles WHERE CONTAINS('unemployment') AND NER(COUNTRY) BIND country AND NER(PERCENT) BIND rate AND DATE(< 2025) BIND date GRANULARITY SENTENCE
# List of unemployment rates joined on current president
SELECT q1.country, q1.rate, q1.date, q2.president, COUNT(q2.president) FROM articles ALIAS q1 JOIN (SELECT date, president FROM wiki WHERE CONTAINS('president') AND NER(COUNTRY, 'united states')  AND DATE(< 2020) BIND date AND NER(PERSON) BIND president GROUP BY president, date GRANULARITY SENTENCE) ALIAS q2 ON q1.date = q2.date WHERE CONTAINS('unemployment') AND NER(COUNTRY, 'united states') BIND country AND NER(PERCENT) BIND rate AND DATE(< 2025) BIND date GROUP BY q1.country, q1.rate, q1.date, q2.president GRANULARITY SENTENCE ORDER BY COUNT(q2.president) DESC
