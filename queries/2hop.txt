# ordinal us pres and party
SELECT q1.ordinal, q1.pres, q2.party FROM source ALIAS q1 JOIN (SELECT party, COUNT(*) FROM source WHERE (CONTAINS('democrat') BIND party OR CONTAINS('republican') BIND party) AND NER(PERSON) BIND person GRANULARITY SENTENCE) ALIAS q2 ON q1.pres = q2.person WHERE DEPENDS('president','amod', '38th') AND CONTAINS('united states') AND NER(ORDINAL) BIND ordinal AND NER(PERSON) BIND pres GRANULARITY SENTENCE

# List of presidents before/after president.
SELECT q2.p, q2.date, COUNT(*) FROM articles ALIAS q1 JOIN (SELECT p, date FROM wiki WHERE CONTAINS('president') AND CONTAINS('united states') AND NER(PERSON) BIND p AND DATE(< 2020) BIND date AND NOT(NER(PERSON, 'reagan')) GROUP BY p, date GRANULARITY SENTENCE) ALIAS q2 ON q1.date BEFORE q2.date WHERE NER(PERSON, 'reagan') BIND p AND CONTAINS('president') AND CONTAINS('united states') AND DATE(< 2050) BIND date GROUP BY q2.date, q2.p GRANULARITY SENTENCE ORDER BY COUNT(*) DESC

# List of unemployment rates joined on current president
SELECT q1.country, q1.rate, q1.date, q2.president, COUNT(q2.president) FROM articles ALIAS q1 JOIN (SELECT date, president FROM wiki WHERE CONTAINS('president') AND NER(COUNTRY, 'united states')  AND DATE(< 2020) BIND date AND NER(PERSON) BIND president GROUP BY president, date GRANULARITY SENTENCE) ALIAS q2 ON q1.date = q2.date WHERE CONTAINS('unemployment') AND NER(COUNTRY, 'united states') BIND country AND NER(PERCENT) BIND rate AND DATE(< 2025) BIND date GROUP BY q1.country, q1.rate, q1.date, q2.president GRANULARITY SENTENCE ORDER BY COUNT(q2.president) DESC

# List of astronauts after Armstrong walked on the moon
SELECT q2.person, q2.date, COUNT(*) FROM source ALIAS q2 JOIN (SELECT date, person FROM source WHERE CONTAINS('moon') AND NER(PERSON, 'Neil Armstrong') BIND person AND DATE(< 2050) BIND date GROUP BY person, date GRANULARITY SENTENCE) ALIAS q1 ON q2.date AFTER q1.date WHERE CONTAINS('astronaut') AND NER(PERSON) BIND person AND DATE(< 2050) BIND date GROUP BY q2.person, q2.date GRANULARITY SENTENCE ORDER BY COUNT(*)
