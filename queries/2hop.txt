# Medical discoveries and researchers with institutions
SELECT q1.researcherName, q1.discoveryDate, q2.organizationName, COUNT(*) FROM articles ALIAS q1 JOIN (SELECT personName, organizationName FROM articles WHERE CONTAINS('hospital') AND NER(ORGANIZATION) BIND organizationName AND NER(PERSON) BIND personName GROUP BY personName, organizationName GRANULARITY SENTENCE) ALIAS q2 ON q1.researcherName = q2.personName WHERE CONTAINS('vaccine') AND NER(PERSON) BIND researcherName AND DATE(INTERSECT [1950,1970]) BIND discoveryDate GROUP BY q1.researcherName, q1.discoveryDate, q2.organizationName GRANULARITY SENTENCE ORDER BY COUNT(*) DESC
