# 3 hop
SELECT q1.country, q1.rate, q1.date, q2.president, COUNT(q2.president) FROM articles ALIAS q1 JOIN ( SELECT date, president FROM wiki WHERE CONTAINS('president') AND NER(COUNTRY, 'united states') AND DATE(< 2020) BIND date AND NER(PERSON) BIND president GROUP BY president, date GRANULARITY SENTENCE ) ALIAS q2 ON q1.date = q2.date JOIN ( SELECT president FROM wiki WHERE CONTAINS('president') AND CONTAINS('Republican Presidential nomination') AND NER(PERSON) BIND president GROUP BY president GRANULARITY SENTENCE ) ALIAS q3 ON q2.president = q3.president WHERE CONTAINS('unemployment') AND NER(COUNTRY, 'united states') BIND country AND NER(PERCENT) BIND rate AND DATE(< 2025) BIND date GROUP BY q1.country, q1.rate, q1.date, q2.president GRANULARITY SENTENCE ORDER BY COUNT(q2.president) DESC

#3hop, nth pres, name, party, death date
SELECT q1.ordinal, q1.pres, q2.party, q3.date, COUNT(q3.date) FROM source ALIAS q1 JOIN (SELECT party, COUNT(*) FROM source WHERE (CONTAINS('democrat') BIND party OR CONTAINS('republican') BIND party) AND NER(PERSON) BIND person GRANULARITY SENTENCE) ALIAS q2 ON q1.pres = q2.person JOIN (SELECT TITLE, person, date FROM source WHERE (CONTAINS('dies')) AND NER(PERSON) BIND person AND DATE(< 2050) BIND date GRANULARITY SENTENCE) ALIAS q3 ON q2.person = q3.person WHERE DEPENDS('president','amod', '38th') AND CONTAINS('united states') AND NER(ORDINAL) BIND ordinal AND NER(PERSON) BIND pres GROUP BY q1.ordinal, q1.pres, q2.party, q3.date GRANULARITY SENTENCE

# 3hop
SELECT q2.person, q3.country FROM source ALIAS q1 JOIN ( SELECT person, date FROM source WHERE CONTAINS('astronaut') AND NER(PERSON) BIND person AND DATE(< 2050) BIND date GROUP BY person, date GRANULARITY SENTENCE ) ALIAS q2 ON q1.date BEFORE q2.date JOIN ( SELECT person, country FROM source WHERE NER(NATIONALITY) BIND country AND NER(PERSON) BIND person GRANULARITY SENTENCE ) ALIAS q3 ON q2.person = q3.person WHERE CONTAINS('moon') AND NER(PERSON, 'Neil Armstrong') BIND person AND DATE(< 2050) BIND date GROUP BY q2.person, q3.country GRANULARITY SENTENCE
