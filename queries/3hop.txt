SELECT q1.country, q1.rate, q1.date, q2.president, COUNT(q2.president) FROM articles ALIAS q1 JOIN ( SELECT date, president FROM wiki WHERE CONTAINS('president') AND NER(COUNTRY, 'united states') AND DATE(< 2020) BIND date AND NER(PERSON) BIND president GROUP BY president, date GRANULARITY SENTENCE ) ALIAS q2 ON q1.date = q2.date JOIN ( SELECT president FROM wiki WHERE CONTAINS('president') AND CONTAINS('Republican Presidential nomination') AND NER(PERSON) BIND president GROUP BY president GRANULARITY SENTENCE ) ALIAS q3 ON q2.president = q3.president WHERE CONTAINS('unemployment') AND NER(COUNTRY, 'united states') BIND country AND NER(PERCENT) BIND rate AND DATE(< 2025) BIND date GROUP BY q1.country, q1.rate, q1.date, q2.president GRANULARITY SENTENCE ORDER BY COUNT(q2.president) DESC

# Explorers & Discoveries
SELECT q1.explorerName, q1.expeditionDate, q2.countryName, q3.exploredLocation, COUNT(*) FROM articles ALIAS q1 JOIN (SELECT personName, countryName FROM articles WHERE NER(NATIONALITY) BIND countryName AND NER(PERSON) BIND personName GROUP BY personName, countryName GRANULARITY SENTENCE) ALIAS q2 ON q1.explorerName = q2.personName JOIN (SELECT personName, exploredLocation FROM articles WHERE (CONTAINS('South Pole') OR CONTAINS('North Pole')) AND NER(LOCATION) BIND exploredLocation AND NER(PERSON) BIND personName GROUP BY personName, exploredLocation GRANULARITY SENTENCE) ALIAS q3 ON q2.personName = q3.personName WHERE (CONTAINS('explorer') OR CONTAINS('expedition')) AND NER(PERSON) BIND explorerName AND DATE(INTERSECT [1925,1950]) BIND expeditionDate GROUP BY q1.explorerName, q1.expeditionDate, q2.countryName, q3.exploredLocation GRANULARITY SENTENCE ORDER BY COUNT(*) DESC

# Political Activists & Organizations
SELECT q1.activistName, q1.activityDate, q2.countryName, q3.movementName, COUNT(*) FROM articles ALIAS q1 JOIN (SELECT personName, countryName FROM articles WHERE NER(NATIONALITY) BIND countryName AND NER(PERSON) BIND personName GROUP BY personName, countryName GRANULARITY SENTENCE) ALIAS q2 ON q1.activistName = q2.personName JOIN (SELECT personName, movementName FROM articles WHERE CONTAINS('civil rights movement') BIND movementName AND NER(PERSON) BIND personName GROUP BY personName, movementName GRANULARITY SENTENCE) ALIAS q3 ON q2.personName = q3.personName WHERE (CONTAINS('activist') OR CONTAINS('leader')) AND NER(PERSON) BIND activistName AND DATE(INTERSECT [1950,1970]) BIND activityDate GROUP BY q1.activistName, q1.activityDate, q2.countryName, q3.movementName GRANULARITY SENTENCE ORDER BY COUNT(*) DESC
SELECT q1.activistName, q1.activityDate, q2.countryName, q3.organizationName, COUNT(*) FROM articles ALIAS q1 JOIN (SELECT personName, countryName FROM articles WHERE NER(NATIONALITY) BIND countryName AND NER(PERSON) BIND personName GROUP BY personName, countryName GRANULARITY SENTENCE) ALIAS q2 ON q1.activistName = q2.personName JOIN (SELECT personName, organizationName FROM articles WHERE CONTAINS('Amnesty International') BIND organizationName AND NER(PERSON) BIND personName GROUP BY personName, organizationName GRANULARITY SENTENCE) ALIAS q3 ON q2.personName = q3.personName WHERE (CONTAINS('human rights') OR CONTAINS('activist')) AND NER(PERSON) BIND activistName AND DATE(INTERSECT [1970,1990]) BIND activityDate GROUP BY q1.activistName, q1.activityDate, q2.countryName, q3.organizationName GRANULARITY SENTENCE ORDER BY COUNT(*) DESC
